cmake_minimum_required(VERSION 3.22)
project(RVOXX VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RVOXX_BUILD_PLUGIN "Build JUCE VST3 and standalone" ON)
set(RVOXX_ARTWORK "${CMAKE_CURRENT_SOURCE_DIR}/Assets/imagem.png" CACHE FILEPATH "Original approved image; embedded without changing its bytes")
include(CTest)
if(BUILD_TESTING)
  add_executable(rvoxx_dsp_tests tests/DynamicsTests.cpp)
  target_include_directories(rvoxx_dsp_tests PRIVATE Source)
  add_test(NAME dynamics COMMAND rvoxx_dsp_tests)
endif()
if(RVOXX_BUILD_PLUGIN)
  if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "RVOXX requires Windows x64. Configure with -A x64.")
  endif()
  if(DEFINED JUCE_DIR)
    find_package(JUCE CONFIG REQUIRED)
  elseif(DEFINED JUCE_PATH)
    add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
      GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
    FetchContent_MakeAvailable(JUCE)
  endif()
  set(rvoxx_manifest TRUE)
  if(CMAKE_CROSSCOMPILING)
    set(rvoxx_manifest FALSE)
  endif()
  juce_add_plugin(RVOXX COMPANY_NAME "RVOXX Project" BUNDLE_ID "audio.rvoxx.plugin"
    PLUGIN_MANUFACTURER_CODE Rvxx PLUGIN_CODE Rvx1
    FORMATS VST3 Standalone PRODUCT_NAME "RVOXX"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics
    VST3_AUTO_MANIFEST ${rvoxx_manifest})
  if(EXISTS "${RVOXX_ARTWORK}")
    juce_add_binary_data(RvoxxArtwork HEADER_NAME RvoxxArtwork.h NAMESPACE RvoxxArtwork
      SOURCES "${RVOXX_ARTWORK}")
    target_link_libraries(RVOXX PRIVATE RvoxxArtwork)
    target_compile_definitions(RVOXX PRIVATE RVOXX_HAS_ARTWORK=1)
  endif()
  target_sources(RVOXX PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
  target_compile_definitions(RVOXX PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(RVOXX PRIVATE juce::juce_audio_utils
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  if(BUILD_TESTING)
    add_executable(rvoxx_processor_tests tests/ProcessorTests.cpp)
    target_include_directories(rvoxx_processor_tests PRIVATE $<TARGET_PROPERTY:RVOXX,INCLUDE_DIRECTORIES>)
    target_compile_definitions(rvoxx_processor_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    target_link_libraries(rvoxx_processor_tests PRIVATE RVOXX)
    add_test(NAME processor COMMAND rvoxx_processor_tests)
  endif()
endif()
